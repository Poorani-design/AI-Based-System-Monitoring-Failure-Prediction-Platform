const si = require("systeminformation");

async function getNetworkSpeed() {
  const stats1 = await si.networkStats();
  
  await new Promise(resolve => setTimeout(resolve, 1000));

  const stats2 = await si.networkStats();

  const rxBytes = stats2[0].rx_bytes - stats1[0].rx_bytes;
  const txBytes = stats2[0].tx_bytes - stats1[0].tx_bytes;

  return {
    download: (rxBytes / 1024).toFixed(2), // KB/s
    upload: (txBytes / 1024).toFixed(2),   // KB/s
  };
}
